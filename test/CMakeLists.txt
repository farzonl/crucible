# Lit/FileCheck-driven test suite.

find_program(DXC_EXECUTABLE dxc
  HINTS "${CMAKE_SOURCE_DIR}/.sdk/bin" /usr/local/bin /opt/homebrew/bin)
if(NOT DXC_EXECUTABLE)
  message(WARNING "dxc not found; tests that compile HLSL will fail")
endif()

if(TARGET offloader)
  set(CRUCIBLE_OFFLOAD_PATH $<TARGET_FILE:offloader>)
  set(CRUCIBLE_API_QUERY_PATH $<TARGET_FILE:api-query>)
endif()

find_program(FILECHECK_EXECUTABLE FileCheck
  HINTS "${LLVM_TOOLS_BINARY_DIR}")
if(NOT FILECHECK_EXECUTABLE)
  message(WARNING "FileCheck not found; lit tests will fail")
endif()

find_program(SPIRV_VAL_EXECUTABLE spirv-val
  HINTS /opt/homebrew/bin /usr/local/bin)
find_program(SPIRV_DIS_EXECUTABLE spirv-dis
  HINTS /opt/homebrew/bin /usr/local/bin)

# lit can come from pip (`pip install lit`), from an LLVM build tree, or be
# pointed at explicitly with -DLLVM_EXTERNAL_LIT=/path/to/lit[.py].
if(NOT LLVM_EXTERNAL_LIT)
  find_program(LLVM_EXTERNAL_LIT NAMES lit llvm-lit
    HINTS "${LLVM_TOOLS_BINARY_DIR}")
endif()

set(CRUCIBLE_CRUCIBLEC_PATH $<TARGET_FILE:cruciblec>)
if(TARGET crucible-run)
  set(CRUCIBLE_RUN_PATH $<TARGET_FILE:crucible-run>)
else()
  set(CRUCIBLE_RUN_PATH "")
endif()
if(TARGET crucible-vk-run)
  set(CRUCIBLE_VK_RUN_PATH $<TARGET_FILE:crucible-vk-run>)
else()
  set(CRUCIBLE_VK_RUN_PATH "")
endif()

set(LIT_SITE_CFG_IN_HEADER
  "# Autogenerated from lit.site.cfg.py.in, do not edit\nimport os")

configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/lit.site.cfg.py.in"
  "${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg.py.gen"
  @ONLY)
file(GENERATE
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg.py"
  INPUT "${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg.py.gen")

if(LLVM_EXTERNAL_LIT)
  set(CRUCIBLE_LIT_COMMAND "${LLVM_EXTERNAL_LIT}")
  if(LLVM_EXTERNAL_LIT MATCHES "\\.py$")
    set(CRUCIBLE_LIT_COMMAND python3 "${LLVM_EXTERNAL_LIT}")
  endif()
  add_custom_target(check-crucible
    COMMAND ${CRUCIBLE_LIT_COMMAND} -sv "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS cruciblec
    USES_TERMINAL
    COMMENT "Running crucible lit tests")
  if(TARGET crucible-run)
    add_dependencies(check-crucible crucible-run)
  endif()
  if(TARGET crucible-vk-run)
    add_dependencies(check-crucible crucible-vk-run)
  endif()

  # llvm/offload-test-suite compliance run (crucible-offload backend).
  # Lives outside this suite's directory so check-crucible doesn't recurse
  # into it.
  if(TARGET offloader)
    add_dependencies(check-crucible offloader api-query)
    if(TARGET imgdiff)
      set(CRUCIBLE_IMGDIFF_PATH $<TARGET_FILE:imgdiff>)
    endif()
    configure_file(
      "${CMAKE_CURRENT_SOURCE_DIR}/offload/lit.site.cfg.py.in"
      "${CMAKE_BINARY_DIR}/test-offload/lit.site.cfg.py.gen"
      @ONLY)
    file(GENERATE
      OUTPUT "${CMAKE_BINARY_DIR}/test-offload/lit.site.cfg.py"
      INPUT "${CMAKE_BINARY_DIR}/test-offload/lit.site.cfg.py.gen")
    add_custom_target(check-crucible-offload-d3d12
      COMMAND ${CRUCIBLE_LIT_COMMAND} -sv
              "${CMAKE_BINARY_DIR}/test-offload"
      DEPENDS offloader api-query
      USES_TERMINAL
      COMMENT "Running llvm/offload-test-suite against Crucible (D3D12 frontend -> Metal/Vulkan)")
    add_custom_target(check-crucible-offload-vk
      COMMAND ${CRUCIBLE_LIT_COMMAND} -sv -Dspirv_frontend=1
              "${CMAKE_BINARY_DIR}/test-offload"
      DEPENDS offloader api-query
      USES_TERMINAL
      COMMENT "Running llvm/offload-test-suite against Crucible (Vulkan frontend -> Metal)")
    add_custom_target(check-crucible-offload-vk-kosmickrisp
      COMMAND ${CRUCIBLE_LIT_COMMAND} -sv -Dspirv_frontend=1
              -Dvulkan_driver=KosmicKrisp "${CMAKE_BINARY_DIR}/test-offload"
      DEPENDS offloader api-query
      USES_TERMINAL
      COMMENT "Running llvm/offload-test-suite against KosmicKrisp")
    add_custom_target(check-crucible-offload-vk-molten
      COMMAND ${CRUCIBLE_LIT_COMMAND} -sv -Dspirv_frontend=1
              -Dvulkan_driver=MoltenVK "${CMAKE_BINARY_DIR}/test-offload"
      DEPENDS offloader api-query
      USES_TERMINAL
      COMMENT "Running llvm/offload-test-suite against MoltenVK")
    add_custom_target(check-crucible-offload-dx
      DEPENDS check-crucible-offload-d3d12)
    add_custom_target(check-crucible-offload-spirv
      DEPENDS check-crucible-offload-vk)
    if(TARGET imgdiff)
      add_dependencies(check-crucible-offload-d3d12 imgdiff)
      add_dependencies(check-crucible-offload-vk imgdiff)
      add_dependencies(check-crucible-offload-vk-kosmickrisp imgdiff)
      add_dependencies(check-crucible-offload-vk-molten imgdiff)
    endif()
  endif()
else()
  message(WARNING
    "lit not found; install with `pip install lit` or pass -DLLVM_EXTERNAL_LIT")
endif()
